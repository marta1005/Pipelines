% ── Analysis ─────────────────────────────────────────────────────────────────
\section{Analysis \& Recommendations}

\begin{itemize}[leftmargin=1.2em,itemsep=2pt,topsep=2pt]
  \item \textbf{Significant failures} (\textit{Frontal gust, Giro}): Error is substantially above target. Investigate whether additional input features explain the residual variance, or consider a more expressive model architecture.
  \item \textbf{Overfitting detected} (\textit{1g, Vertical gust, Turn, n0, Giro}): KS test rejects distributional equality between train and test residuals. Apply stronger regularisation (dropout, weight decay) or reduce model complexity.
\end{itemize}
